/* 广播排程主表 */
CREATE TABLE IF NOT EXISTS pa_broadcast_schedule
(
    schedule_id           BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    location_id           INT NOT NULL,
    schedule_name         VARCHAR(100) NOT NULL,
    enabled               TINYINT(1) NOT NULL DEFAULT 0,

    msg_id                INT NOT NULL,
    msg_version           VARCHAR(50) NOT NULL,

    /* 多个车站编号，例如：1,3,12 */
    stations              VARCHAR(1000) NOT NULL,

    /* 区域位掩码，例如：5 = 二进制 00000101 */
    zones                 INT UNSIGNED NOT NULL,

    seat_id               INT NOT NULL,
    language              TINYINT UNSIGNED NOT NULL DEFAULT 1,
    play_count            TINYINT UNSIGNED NOT NULL DEFAULT 1,
    play_interval_sec     TINYINT UNSIGNED NOT NULL DEFAULT 0,

    schedule_type         ENUM('ONCE', 'DAILY', 'WEEKLY') NOT NULL,
    start_at              DATETIME(3) NOT NULL,
    end_at                DATETIME(3) NULL,

    /* 每 N 天或每 N 周执行一次 */
    repeat_interval       SMALLINT UNSIGNED NOT NULL DEFAULT 1,

    /* 周排程使用：bit 0~6 分别表示周一~周日 */
    weekday_mask          TINYINT UNSIGNED NULL,

    /* 调度服务计算和维护 */
    next_run_at           DATETIME(3) NULL,
    last_run_at           DATETIME(3) NULL,

    created_by            VARCHAR(100) NULL,
    created_at            DATETIME(3) NOT NULL DEFAULT CURRENT_TIMESTAMP(3),
    updated_by            VARCHAR(100) NULL,
    updated_at            DATETIME(3) NOT NULL DEFAULT CURRENT_TIMESTAMP(3)
                                           ON UPDATE CURRENT_TIMESTAMP(3),

    PRIMARY KEY (schedule_id),

    KEY idx_pa_bc_schedule_due (enabled, next_run_at),
    KEY idx_pa_bc_schedule_message (msg_id, msg_version),

    CONSTRAINT chk_pa_bc_schedule_enabled
        CHECK (enabled IN (0, 1)),

    CONSTRAINT chk_pa_bc_schedule_play_count
        CHECK (play_count BETWEEN 1 AND 255),

    CONSTRAINT chk_pa_bc_schedule_end_at
        CHECK (end_at IS NULL OR end_at >= start_at),

    CONSTRAINT chk_pa_bc_schedule_repeat_interval
        CHECK (repeat_interval >= 1),

    CONSTRAINT chk_pa_bc_schedule_weekday_mask
        CHECK (
            schedule_type <> 'WEEKLY'
            OR weekday_mask BETWEEN 1 AND 127
        )
)
ENGINE = InnoDB
DEFAULT CHARSET = utf8mb4
COLLATE = utf8mb4_unicode_ci;


/* 排程执行日志表 */
CREATE TABLE IF NOT EXISTS pa_broadcast_schedule_log
(
    run_id                BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    schedule_id           BIGINT UNSIGNED NOT NULL,

    planned_at            DATETIME(3) NOT NULL,
    started_at            DATETIME(3) NULL,
    finished_at           DATETIME(3) NULL,

    status                ENUM(
                              'WAITING',
                              'IN_PROGRESS',
                              'SUCCESS',
                              'PARTIAL_FAILED',
                              'FAILED',
                              'SKIPPED',
                              'CANCELLED'
                          ) NOT NULL,

    announce_id           INT NULL,

    /* 记录执行时的配置快照 */
    msg_id                INT NOT NULL,
    msg_version           VARCHAR(50) NOT NULL,
    stations              VARCHAR(1000) NOT NULL,
    zones                 INT UNSIGNED NOT NULL,

    error_code            VARCHAR(100) NULL,
    error_message         VARCHAR(2000) NULL,
    triggered_by          ENUM('SCHEDULER', 'MANUAL')
                          NOT NULL DEFAULT 'SCHEDULER',

    created_at            DATETIME(3) NOT NULL DEFAULT CURRENT_TIMESTAMP(3),

    PRIMARY KEY (run_id),

    KEY idx_pa_bc_schedule_log_schedule (schedule_id, planned_at),

    CONSTRAINT fk_pa_bc_schedule_log_schedule
        FOREIGN KEY (schedule_id)
        REFERENCES pa_broadcast_schedule (schedule_id)
)
ENGINE = InnoDB
DEFAULT CHARSET = utf8mb4
COLLATE = utf8mb4_unicode_ci;
